cmake_minimum_required(VERSION 3.16)
project(chassis_controllers)
find_package(ament_cmake REQUIRED)
add_library(chassis_kinematics INTERFACE)
target_compile_features(chassis_kinematics INTERFACE cxx_std_17)
target_include_directories(chassis_kinematics INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
install(DIRECTORY include/ DESTINATION include)
install(TARGETS chassis_kinematics EXPORT export_chassis)
ament_export_targets(export_chassis HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  enable_testing()
  add_executable(kinematics_check test/kinematics_check.cpp)
  target_link_libraries(kinematics_check chassis_kinematics)
  add_test(NAME kinematics_check COMMAND kinematics_check)
endif()
ament_package()
