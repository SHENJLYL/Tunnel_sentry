cmake_minimum_required(VERSION 3.16)
project(universal_controller)
find_package(ament_cmake REQUIRED)
foreach(dependency chassis_controllers controller_interface pluginlib rclcpp realtime_tools geometry_msgs std_msgs sensor_msgs)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(sentry_controller SHARED src/sentry_controller.cpp)
target_compile_features(sentry_controller PUBLIC cxx_std_17)
target_compile_options(sentry_controller PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(sentry_controller chassis_controllers controller_interface pluginlib rclcpp realtime_tools geometry_msgs std_msgs sensor_msgs)
pluginlib_export_plugin_description_file(controller_interface sentry_plugins.xml)
install(TARGETS sentry_controller LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
